import re, sys
orig = open('/tmp/chart-original.html').read()
new  = open('chart.html').read()

def bar_attrs(s):
    out = []
    for tag in re.findall(r'<rect\b[^>]*>', s):
        if 'data-value' not in tag: continue
        out.append(dict(re.findall(r'\b(data-[a-z]+|x|y|width|height)="([^"]*)"', tag)))
    return out

bo, bn = bar_attrs(orig), bar_attrs(new)
errs = []
if bo != bn:
    errs.append('bar rects changed')
    for a, b in zip(bo, bn):
        if a != b: errs.append(f'{a} -> {b}')

ticks_o = re.findall(r'<text[^>]*text-anchor="end"[^>]*>([^<]+)</text>', orig)
ticks_n = re.findall(r'<text[^>]*text-anchor="end"[^>]*>([^<]+)</text>', new)
if ticks_o != ticks_n:

vals = [d['data-value'] for d in bn]
labels = re.findall(r'<text class="value-label"[^>]*>([^<]+)</text>', new)
if labels != vals:

cats_o = re.findall(r'data-cat="([^"]+)"', orig)
cats_n = re.findall(r'data-cat="([^"]+)"', new)
ser_o  = re.findall(r'data-series="([^"]+)"', orig)
ser_n  = re.findall(r'data-series="([^"]+)"', new)
if (cats_o, ser_o) != (cats_n, ser_n):

if errs:
    print('\n'.join(errs)); sys.exit(1)
print('OK: all 12 bar geometries, values, categories, series and axis tick labels unchanged')
